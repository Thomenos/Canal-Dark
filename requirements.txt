# --- EDIÇÃO DE VÍDEO ---
moviepy==1.0.3
# Biblioteca principal para edição de vídeo
# Versão 1.0.3 é a mais estável para vídeos longos

pillow<10.0.0
# Processamento de imagens (dependência do MoviePy)
# Versões 10+ causam incompatibilidade com MoviePy 1.0.3

opencv-python
# Processamento de vídeo em tempo real
# Usado para o efeito de pulso/respiração no background

# --- SÍNTESE DE VOZ ---
edge-tts
# Geração de narração com vozes neurais da Microsoft Edge
# Gratuito e de alta qualidade (voz Brian Multilingual)

# --- GERAÇÃO DE ROTEIROS (OPCIONAL) ---
google-generativeai
# API do Google Gemini para gerar histórias de terror
# Requer chave API (configure CHAVE_GEMINI no app.py)

# --- UPLOAD PARA YOUTUBE ---
google-api-python-client
# API oficial do YouTube para upload de vídeos
google-auth-oauthlib
# Autenticação OAuth2 para YouTube
google-auth-httplib2
# Suporte HTTP para Google Auth

# --- DOWNLOAD DE IMAGENS ---
requests
# Biblioteca para fazer requisições HTTP
# Usada para baixar backgrounds da API Pollinations

# --- DEPENDÊNCIAS ADICIONAIS (instaladas automaticamente) ---
# imageio-ffmpeg (para codecs de vídeo)
# numpy (processamento numérico)
# decorator (utilitários MoviePy)
# proglog (barra de progresso da renderização)
